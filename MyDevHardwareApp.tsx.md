
import React, { useState, useCallback, useEffect, useRef } from 'react';
import { DeviceData, SearchHistoryItem } from './types';
import { fetchDeviceIntelligence } from './services/geminiService';
import { DeviceDashboard } from './components/DeviceDashboard';

const App: React.FC = () => {
  const [model, setModel] = useState('');
  const [os, setOs] = useState('');
  const [isLoading, setIsLoading] = useState(false);
  const [statusMessage, setStatusMessage] = useState('');
  const [elapsedTime, setElapsedTime] = useState(0);
  const [estimatedTotal] = useState(25); // Estimated 25 seconds for deep audit
  const [currentDevice, setCurrentDevice] = useState<DeviceData | null>(null);
  const [history, setHistory] = useState<SearchHistoryItem[]>([]);
  const [error, setError] = useState<string | null>(null);
  
  const timerRef = useRef<number | null>(null);

  const handleKeySelection = async () => {
    try {
      const aiStudio = (window as any).aistudio;
      if (aiStudio?.openSelectKey) {
        await aiStudio.openSelectKey();
        setError(null);
      }
    } catch (e) {
      console.error("Key selection failed:", e);
    }
  };

  const generateMarkdownReport = (data: DeviceData): string => {
    const { insights, manufacturer, model, os, timestamp, specs, groundingSources } = data;
    let md = `# ENGINEERING AUDIT REPORT: ${manufacturer.toUpperCase()} ${model.toUpperCase()}\n`;
    md += `**Generated By:** RepairIntel Hub PRO v4.5\n`;
    md += `**OS Deployment Target:** ${os}\n`;
    md += `**Audit Timestamp:** ${timestamp}\n`;
    md += `**Status:** GROUNDING VERIFIED\n\n`;
    
    md += `## [1] HARDWARE ARCHITECTURE\n`;
    md += `| Component | Specification |\n`;
    md += `| :--- | :--- |\n`;
    md += `| Processor | ${specs?.processor || 'N/A'} |\n`;
    md += `| Memory | ${specs?.ram || 'N/A'} |\n`;
    md += `| Storage | ${specs?.storage || 'N/A'} |\n`;
    md += `| Display | ${specs?.display || 'N/A'} |\n`;
    md += `| Power/Battery | ${specs?.battery || 'N/A'} |\n\n`;

    if (insights.hazardsAndRecalls.length > 0) {
      md += `## [2] SAFETY & RECALL REGISTRY\n`;
      insights.hazardsAndRecalls.forEach(h => {
        md += `### [${h.severity}] ${h.title}\n`;
        md += `- **Category:** ${h.category}\n`;
        md += `- **Detail:** ${h.description}\n`;
        md += `- **Required Action:** ${h.actionRequired}\n`;
        if (h.sourceUrl) md += `- **Link:** ${h.sourceUrl}\n`;
        md += `\n`;
      });
    }

    md += `## [3] AUTOMATION SCRIPTS\n`;
    md += `### WINDOWS ADMIN CMD\n\`\`\`batch\n${insights.automationScripts.cmd}\n\`\`\`\n\n`;
    md += `### WSL UBUNTU / LINUX\n\`\`\`bash\n${insights.automationScripts.wsl}\n\`\`\`\n\n`;
    md += `### TERMUX ANDROID\n\`\`\`bash\n${insights.automationScripts.termux}\n\`\`\`\n\n`;

    if (groundingSources && groundingSources.length > 0) {
      md += `## [4] INTELLIGENCE SOURCES\n`;
      groundingSources.forEach(s => {
        md += `- [${s.title}](${s.url})\n`;
      });
    }

    return md;
  };

  const exportData = useCallback((data: DeviceData) => {
    try {
      const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
      const baseName = `${data.manufacturer}_${data.model.replace(/\s+/g, '_')}_${timestamp}`;
      
      const jsonBlob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
      const jsonLink = document.createElement('a');
      jsonLink.href = URL.createObjectURL(jsonBlob);
      jsonLink.download = `VAULT_${baseName}.json`;
      jsonLink.click();

      const mdContent = generateMarkdownReport(data);
      const mdBlob = new Blob([mdContent], { type: 'text/markdown' });
      const mdLink = document.createElement('a');
      mdLink.href = URL.createObjectURL(mdBlob);
      mdLink.download = `REPORT_${baseName}.md`;
      mdLink.click();
    } catch (err) {
      console.error("Critical Export Error:", err);
    }
  }, []);

  const handleSearch = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!model || !os) return;
    
    setIsLoading(true);
    setError(null);
    setElapsedTime(0);
    setStatusMessage("Initializing Intelligence Grid...");
    
    // Telemetry and status timer
    timerRef.current = window.setInterval(() => {
      setElapsedTime(prev => prev + 0.1);
      
      setStatusMessage(prev => {
        if (elapsedTime > 20) return "Finalizing Forensic Data Streams...";
        if (elapsedTime > 15) return "Synthesizing Cross-Platform Scripts...";
        if (elapsedTime > 10) return "Analyzing Manufacturing Schematics...";
        if (elapsedTime > 5) return "Scrubbing Recall Databases...";
        return prev;
      });
    }, 100);

    try {
      const data = await fetchDeviceIntelligence(model, os);
      setCurrentDevice(data);
      setHistory(prev => [{ 
        id: crypto.randomUUID(), 
        model: data.model, 
        manufacturer: data.manufacturer, 
        timestamp: data.timestamp, 
        data 
      }, ...prev].slice(0, 50));
      
      // Automatic download of all generated data
      exportData(data);
    } catch (err: any) {
      let msg = err.message || 'Audit Interrupted.';
      if (msg.includes('403') || msg.includes('permission') || msg.includes('not found')) {
        msg = "Authorization Fault: Select a valid API key for forensic access.";
      }
      setError(msg);
    } finally {
      if (timerRef.current) clearInterval(timerRef.current);
      setIsLoading(false);
    }
  };

  const progress = Math.min(99, (elapsedTime / estimatedTotal) * 100);
  const remainingTime = Math.max(0, estimatedTotal - elapsedTime);

  return (
    <div className="min-h-screen flex flex-col bg-[#0a0c10] text-slate-100 selection:bg-indigo-500/30">
      <nav className="bg-[#0f1117] sticky top-0 z-50 px-8 py-4 border-b border-white/5 shadow-2xl backdrop-blur-md">
        <div className="max-w-[1800px] mx-auto flex items-center justify-between">
          <div className="flex items-center space-x-6">
            <div className="w-12 h-12 bg-indigo-600 rounded-xl flex items-center justify-center font-black text-2xl shadow-lg shadow-indigo-500/20">RI</div>
            <div>
              <h1 className="text-xl font-black tracking-tighter leading-none text-white">RepairIntel Hub</h1>
              <p className="text-[9px] text-indigo-400 uppercase font-black tracking-[0.4em] mt-1">Forensic Suite v4.5 PRO</p>
            </div>
          </div>
          <div className="flex items-center space-x-6">
             <button 
               onClick={handleKeySelection}
               className="px-4 py-2 bg-slate-800 hover:bg-slate-700 border border-slate-700 rounded-lg text-[9px] font-black uppercase tracking-widest text-slate-300 transition-all"
             >
               Engine Config
             </button>
             <div className="hidden md:flex items-center space-x-3 bg-black/40 px-4 py-2 rounded-full border border-white/5">
                <span className="w-2 h-2 rounded-full bg-emerald-500 animate-pulse"></span>
                <span className="text-[9px] font-black uppercase tracking-widest text-slate-400">Grid Connected</span>
             </div>
          </div>
        </div>
      </nav>

      <main className="flex-grow max-w-[1800px] mx-auto w-full p-6 md:p-10">
        {/* Search Matrix */}
        <div className="mb-10">
          <div className="bg-[#14171f] border border-white/5 rounded-[32px] p-8 shadow-2xl">
            <h2 className="text-xs font-black text-indigo-400 mb-6 uppercase tracking-[0.3em] flex items-center">
              Target Acquisition Matrix
              <span className="ml-4 h-[1px] flex-grow bg-white/5"></span>
            </h2>
            <form onSubmit={handleSearch} className="grid grid-cols-1 md:grid-cols-3 gap-6 items-end">
              <div className="space-y-2">
                <label className="text-[10px] font-black text-slate-500 uppercase tracking-widest ml-1">Device Model Identifier</label>
                <input type="text" value={model} onChange={(e) => setModel(e.target.value)} placeholder="e.g. HP EliteBook 840 G8" className="w-full px-6 py-4 bg-black/40 border border-white/10 rounded-xl outline-none text-sm font-bold text-white focus:border-indigo-500/50 transition-all placeholder:text-slate-700" required />
              </div>
              <div className="space-y-2">
                <label className="text-[10px] font-black text-slate-500 uppercase tracking-widest ml-1">Environment OS</label>
                <input type="text" value={os} onChange={(e) => setOs(e.target.value)} placeholder="e.g. Ubuntu 22.04 LTS" className="w-full px-6 py-4 bg-black/40 border border-white/10 rounded-xl outline-none text-sm font-bold text-white focus:border-indigo-500/50 transition-all placeholder:text-slate-700" required />
              </div>
              <button 
                type="submit" 
                disabled={isLoading} 
                className="font-black py-4 rounded-xl shadow-xl transition-all disabled:opacity-50 text-[11px] uppercase tracking-[0.2em] text-white bg-indigo-600 hover:bg-indigo-500 active:scale-[0.98]"
              >
                {isLoading ? 'EXECUTING AUDIT...' : 'Run Forensic Audit'}
              </button>
            </form>
          </div>
        </div>

        {/* Dynamic Display Area */}
        <div className="w-full min-h-[500px]">
          {error && (
            <div className="bg-rose-500/10 border border-rose-500/20 text-rose-200 p-8 rounded-[32px] mb-10 flex items-start animate-in zoom-in duration-300">
              <span className="text-3xl mr-6">🚫</span>
              <div className="flex-grow">
                <p className="font-black text-[14px] uppercase tracking-wider">Telemetry Interrupt Detected</p>
                <p className="text-xs mt-2 font-medium opacity-80 leading-relaxed">{error}</p>
                <div className="mt-6 flex space-x-3">
                   <button onClick={handleSearch} className="px-5 py-2.5 bg-rose-600 text-white rounded-lg text-[9px] font-black uppercase tracking-widest hover:bg-rose-700 transition-colors">Restart Audit</button>
                </div>
              </div>
            </div>
          )}
          
          {isLoading ? (
            <div className="h-[500px] flex flex-col items-center justify-center bg-[#14171f] rounded-[48px] border border-white/5 p-12 space-y-12 animate-in fade-in duration-500 relative overflow-hidden">
              {/* Scanline Effect */}
              <div className="absolute inset-0 pointer-events-none bg-gradient-to-b from-transparent via-indigo-500/5 to-transparent h-full w-full animate-scanline"></div>
              
              <div className="relative">
                <div className="w-40 h-40 border-[4px] border-white/5 rounded-full"></div>
                <div className="absolute top-0 w-40 h-40 border-[4px] border-indigo-500 border-t-transparent rounded-full animate-spin"></div>
                <div className="absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 flex flex-col items-center">
                   <span className="text-3xl mb-1">🛰️</span>
                   <span className="text-[10px] font-black text-indigo-400 animate-pulse uppercase tracking-[0.2em]">Live Data</span>
                </div>
              </div>

              <div className="text-center w-full max-w-2xl space-y-8 relative z-10">
                <div className="space-y-2">
                  <h2 className="text-2xl font-black text-white tracking-tighter uppercase">{statusMessage}</h2>
                  <p className="text-slate-500 font-bold text-[10px] uppercase tracking-[0.3em]">Hardware Analysis Engine In Progress</p>
                </div>

                {/* Precision Progress System */}
                <div className="space-y-4">
                  <div className="flex justify-between items-end mb-2">
                    <div className="text-left">
                       <p className="text-[10px] font-black text-slate-500 uppercase tracking-widest">Elapsed Telemetry</p>
                       <p className="text-sm font-mono text-indigo-400">{elapsedTime.toFixed(1)}s</p>
                    </div>
                    <div className="text-center">
                       <p className="text-[10px] font-black text-slate-500 uppercase tracking-widest">Completion</p>
                       <p className="text-2xl font-black text-white">{Math.round(progress)}%</p>
                    </div>
                    <div className="text-right">
                       <p className="text-[10px] font-black text-slate-500 uppercase tracking-widest">Remaining (EST)</p>
                       <p className="text-sm font-mono text-indigo-400">~{remainingTime.toFixed(1)}s</p>
                    </div>
                  </div>
                  <div className="h-4 bg-black/40 rounded-full border border-white/5 p-1 overflow-hidden">
                    <div 
                      className="h-full bg-gradient-to-r from-indigo-600 via-violet-500 to-indigo-400 rounded-full shadow-[0_0_15px_rgba(99,102,241,0.5)] transition-all duration-300 ease-out relative"
                      style={{ width: `${progress}%` }}
                    >
                      <div className="absolute top-0 right-0 h-full w-2 bg-white/40 blur-[2px]"></div>
                    </div>
                  </div>
                </div>
              </div>
            </div>
          ) : currentDevice ? (
            <DeviceDashboard 
              data={currentDevice} 
              history={history}
              onExport={exportData}
              onSelectAudit={setCurrentDevice}
            />
          ) : (
            <div className="h-[500px] flex flex-col items-center justify-center text-center space-y-8 bg-[#14171f] border border-dashed border-white/10 rounded-[48px] p-10">
               <div className="w-32 h-32 bg-black/40 rounded-[32px] border border-white/5 flex items-center justify-center text-5xl grayscale opacity-50">🛠️</div>
               <div className="space-y-3">
                 <h2 className="text-3xl font-black text-white tracking-tight">System Idle.</h2>
                 <p className="text-slate-500 font-bold text-xs max-w-sm mx-auto leading-relaxed">The Forensic Intelligence Hub is ready. Input hardware identifiers to begin deep system synthesis.</p>
               </div>
            </div>
          )}
        </div>
      </main>

      {/* Global CSS for the scanline */}
      <style>{`
        @keyframes scanline {
          0% { transform: translateY(-100%); opacity: 0; }
          50% { opacity: 1; }
          100% { transform: translateY(100%); opacity: 0; }
        }
        .animate-scanline {
          animation: scanline 4s linear infinite;
        }
      `}</style>
    </div>
  );
};

export default App;
